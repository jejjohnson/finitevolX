    pipekit Operators

    finitevolx.pipekit wraps the 2-D Cartesian operators as pipekit operators: JSON configuration, Sequential / Graph composition, and Equinox pytree leaves so jit, grad and vmap see through them. Each operator is one Tier-A call on [*batch, Ny, Nx] fields. Requires the pipekit extra: pip install "finitevolx[pipekit]". Walkthrough: Composing finitevolX with pipekit.

    Tier-B operator Tier-A counterpart in → out
    Difference(axis, src, dst) Difference2D.diff_{axis}_{src}_to_{dst} 1 → 1
    Divergence Difference2D.divergence (U, V) → T
    Curl Difference2D.curl (U, V) → X
    Laplacian Difference2D.laplacian T → T
    GradPerp Difference2D.grad_perp X → (U, V)
    Interp(src, dst) Interpolation2D.{src}_to_{dst} 1 → 1
    Coriolis(f0, beta, y0) coriolis_fn + Coriolis2D (U, V) → (U, V)
    FillGhosts(kind) enforce_periodic, zero_boundaries, …, wall_boundaries 1 → 1
    ApplyMask(mask, stagger) jnp.where(mask.{h,u,v,xy_corner_strict}, x, 0) 1 → 1
    RelativeVorticity Vorticity2D.relative_vorticity (U, V) → X
    PotentialVorticity Vorticity2D.potential_vorticity (U, V, T, T) → X
    Enstrophy Vorticity2D.enstrophy (U, V) → X
    QGPotentialVorticity(f0, beta, y0) QGPotentialVorticity2D T → T
    KineticEnergy Energetics2D.kinetic_energy (U, V) → T
    AvailablePE(g_prime, H) Energetics2D.available_potential_energy T → T
    Strain(component) Strain2D.shear / .tensor / .magnitude_squared (U, V) → X / T
    OkuboWeiss Strain2D.okubo_weiss (U, V) → T
    TotalEnergy total_energy (T, T) → [*batch]
    DomainIntegral / DomainMean area_sum / area_mean T → [*batch]
    ExplicitStep(rhs, dt, scheme) euler_step / heun_step / rk3_ssp_step / rk4_step state → state
    Pick(index), LinComb(coeffs) — (graph plumbing)

    def as_grid_spec(spec: Sequence[Any] | CartesianGrid2D) -> GridSpec:
    """Normalise a grid description to a `GridSpec` tuple.

    Parameters
    ----------
    spec : sequence of 4 numbers or CartesianGrid2D
        ``(nx_interior, ny_interior, Lx, Ly)`` (a list after a JSON round
        trip), or a grid to describe.

    Returns
    -------
    GridSpec
        ``(int, int, float, float)``, hashable, so it can be a static field.

    Raises
    ------
    ValueError
        If ``spec`` does not have four entries or a count is not positive.
    """
    if isinstance(spec, CartesianGrid2D):
        spec = (spec.Nx - 2, spec.Ny - 2, spec.Lx, spec.Ly)
    if len(spec) != 4:
        raise ValueError(
            f"grid spec must be (nx_interior, ny_interior, Lx, Ly), got {spec!r}"
        )
    nx, ny, lx, ly = spec
    if int(nx) < 1 or int(ny) < 1:
        raise ValueError(f"grid spec needs positive cell counts, got {spec!r}")
    return (int(nx), int(ny), float(lx), float(ly))

    def over_leading(fn: Callable[..., Any], core_ndim: int = 2) -> Callable[..., Any]:
    """Lift ``fn`` on ``[*core]`` arrays to ``[*batch, *core]`` arrays.

    Every input shares the same leading shape; it is flattened to one axis,
    ``jax.vmap(fn)`` maps over it, and every output array gets the leading
    shape back::

        x : [T, E, Ny, Nx] ─reshape─► [T·E, Ny, Nx] ─vmap fn─► [T·E, …] ─reshape─► [T, E, …]

    Only needed for Tier-A functions that are still 2-D only; the 2-D
    Cartesian operators take ``[*batch, Ny, Nx]`` natively.

    Parameters
    ----------
    fn : callable
        Function of positional arrays with ``core_ndim`` trailing axes each;
        it may return an array or a pytree of arrays.
    core_ndim : int
        Number of trailing axes ``fn`` consumes (2 for ``[Ny, Nx]``).

    Returns
    -------
    callable
        ``fn`` lifted over any number of leading axes (zero included).

    Raises
    ------
    ValueError
        If the inputs disagree on their leading shape.
    """

    @functools.wraps(fn)
    def lifted(*xs: Array) -> Any:
        leads = {tuple(x.shape[: x.ndim - core_ndim]) for x in xs}
        if len(leads) != 1:
            raise ValueError(f"inputs have different leading shapes: {sorted(leads)}")
        (lead,) = leads
        if not lead:
            return fn(*xs)
        flat = [x.reshape(-1, *x.shape[x.ndim - core_ndim :]) for x in xs]
        out = jax.vmap(fn)(*flat)
        return jax.tree.map(lambda y: y.reshape(*lead, *y.shape[1:]), out)

    return lifted

    def check_operator(op: Any, *fields: Any, hlo: bool = True) -> None:
    """Assert that ``op`` is a well-behaved finitevolX pipekit operator.

    Runs `pipekit_jax.testing.check_jax_operator` (operator + module, eager
    ≡ ``filter_jit``, ``vmap``, visible leaves, JSON config round trip),
    then checks:

    1. **Leading axes**: on inputs ``[*lead, Ny, Nx]`` with ``lead`` = ``(2,)``
       and ``(2, 3)``, ``op`` equals a loop over the members.
    2. **Gradient**: ``∂/∂x₀ ∑ op(x)`` is finite.
    3. **Lowering** (when ``hlo``): the compiled HLO has no ``gather``,
       ``scatter`` or ``convolution``.

    Parameters
    ----------
    op : FVXOperator
        Operator under test.
    *fields : Array
        Example inputs ``[Ny, Nx]`` (non-array inputs are passed through).
    hlo : bool
        Also check the lowering.

    Raises
    ------
    AssertionError
        On the first violated property.
    """
    check_jax_operator(op, *fields)

    for lead in ((2,), (2, 3)):
        members = _member_inputs(fields, lead)
        per = [op(*m) for m in members]
        want = jax.tree.map(
            lambda *ys, lead=lead: jnp.stack(ys).reshape(*lead, *ys[0].shape), *per
        )
        batched = tuple(
            jnp.stack([m[i] for m in members]).reshape(*lead, *f.shape)
            if eqx.is_array(f)
            else f
            for i, f in enumerate(fields)
        )
        got = op(*batched)
        assert jax.tree.structure(got) == jax.tree.structure(want), (
            f"leading axes {lead}: output structure differs"
        )
        for a, b in zip(jax.tree.leaves(got), jax.tree.leaves(want), strict=True):
            # Eager batched ≡ per-member up to rounding (a few ulp).
            tol = 64 * float(jnp.finfo(a.dtype).eps)
            np.testing.assert_allclose(
                a, b, rtol=tol, atol=tol, err_msg=f"leading axes {lead}"
            )

    first = next(i for i, f in enumerate(fields) if eqx.is_inexact_array(f))

    def loss(x: Any) -> Any:
        xs = list(fields)
        xs[first] = x
        return sum(jnp.sum(jnp.real(y)) for y in jax.tree.leaves(op(*xs)))

    grad = jax.grad(loss)(fields[first])
    assert bool(jnp.all(jnp.isfinite(grad))), "gradient w.r.t. the first input"

    if hlo:
        text = compiled_hlo(op, *fields)
        found = [name.rstrip("(") for name in FORBIDDEN_HLO if name in text]
        assert not found, f"compiled HLO contains {found}"

    class LinComb(FVXOperator):
    """Linear combination y = ∑ᵢ cᵢ xᵢ (coefficients are learnable leaves).

    Attributes
    ----------
    coeffs : Array
        ``[n]`` coefficients, one per input.
    """

    coeffs: jax.Array = eqx.field(converter=jnp.asarray)

    def _apply(self, *xs: Field) -> Field:
        if len(xs) != self.coeffs.shape[0]:
            raise ValueError(
                f"LinComb has {self.coeffs.shape[0]} coefficients, got {len(xs)} inputs"
            )
        return sum((c * x for c, x in zip(self.coeffs, xs, strict=True)), start=0.0)
