    Analysis kernels (gridax)

    Finite differences and remapping on collocated grids, from gridax.analysis. They compute in the namespace of their input, so the same call runs on NumPy, on JAX (also under jax.jit), or on any backend array-api-compat knows, and returns the input's dtype. They are the kernels behind geoops' and xrops' Gradient, Divergence, Curl, Laplacian and vertical-remap operators. For fields on the staggered C-grid, use the Difference* operators instead.

    gridax.analysis

    Analysis kernels on collocated grids: finite differences and remapping.

    Every function here computes in the namespace of its input, so it runs on NumPy, JAX (under jit too), CuPy, or any backend array-api-compat knows, and returns the input's dtype. They are the kernels behind geoops' and xrops' Gradient, Divergence, Curl, Laplacian and vertical-remap operators (#423).

    • Stencils (L0): :func:difference_offsets, :func:difference_coeffs, :func:nonuniform_coeffs, from integers alone.
    • Differences (L1): :func:difference (a NaN propagates) and :func:adaptive_difference (the widest stencil that fits on valid data), and the 1-D / 2-D / 3-D gradient, divergence, curl and Laplacian built on them, each with an adaptive_ twin.
    • Remapping: :func:remap_axis (one 1-D source coordinate) and :func:remap_axis_columns (one set of levels per column).

    def adaptive_curl_2d(
    u: Float[np.ndarray, "*batch h w"],
    v: Float[np.ndarray, "*batch h w"],
    dy: Float[np.ndarray, "h w"] | float = 1.0,
    dx: Float[np.ndarray, "h w"] | float = 1.0,
    *,
    mask: Bool[np.ndarray, "*batch h w"] | None = None,
    accuracy: int = 1,
    periodic_x: bool = False,
) -> Float[np.ndarray, "*batch h w"]:
    """Mask-aware curl: :func:`adaptive_difference` for each derivative.

    Args:
        u: The component along columns.
        v: The component along rows.
        dy: The signed step between row centres.
        dx: The signed step between column centres.
        mask: ``True`` drops the cell; ``None`` drops only non-finite cells.
        accuracy: The highest accuracy order to try.
        periodic_x: The columns wrap.

    Returns:
        The curl, ``u``'s shape, ``NaN`` only where a cell is dropped.

    Example:
        >>> y, x = np.mgrid[0:5, 0:6].astype(float)
        >>> wall = x == 2
        >>> zeta = adaptive_curl_2d(-y, x, mask=wall)
        >>> int(np.isnan(zeta).sum()), float(np.nanmax(np.abs(zeta - 2.0)))
        (5, 0.0)
    """
    d_v = adaptive_difference(
        v, mask=mask, axis=-1, step=dx, accuracy=accuracy, periodic=periodic_x
    )
    return d_v - adaptive_difference(u, mask=mask, axis=-2, step=dy, accuracy=accuracy)

    def adaptive_curl_3d(
    u: Float[np.ndarray, "*batch d h w"],
    v: Float[np.ndarray, "*batch d h w"],
    w: Float[np.ndarray, "*batch d h w"],
    dz: Float[np.ndarray, "d h w"] | float = 1.0,
    dy: Float[np.ndarray, "h w"] | float = 1.0,
    dx: Float[np.ndarray, "h w"] | float = 1.0,
    *,
    z: Float[np.ndarray, " d"] | None = None,
    mask: Bool[np.ndarray, "*batch d h w"] | None = None,
    accuracy: int = 1,
    periodic_x: bool = False,
) -> Float[np.ndarray, "3 *batch d h w"]:
    """Mask-aware 3-D curl: :func:`adaptive_difference` for each derivative.

    Args:
        u: The component along columns.
        v: The component along rows.
        w: The component along levels.
        dz: The signed step between level centres.
        dy: The signed step between row centres.
        dx: The signed step between column centres.
        z: The level centres' positions; replaces ``dz``.
        mask: ``True`` drops the cell; ``None`` drops only non-finite cells.
        accuracy: The highest accuracy order to try.
        periodic_x: The columns wrap.

    Returns:
        ``(ω_z, ω_y, ω_x)`` stacked on a new leading axis.

    Example:
        >>> z, y, x = np.mgrid[0:4, 0:5, 0:6].astype(float)
        >>> rock = (z == 2) & (y == 2) & (x == 3)
        >>> zeta_z = adaptive_curl_3d(-y, x, 0.0 * z, mask=rock)[0]
        >>> int(np.isnan(zeta_z).sum()), float(np.nanmax(np.abs(zeta_z - 2.0)))
        (1, 0.0)
    """
    xp = array_namespace(u)
    kw: dict[str, Any] = {"mask": mask, "accuracy": accuracy}
    d_x = functools.partial(
        adaptive_difference, axis=-1, step=dx, periodic=periodic_x, **kw
    )
    d_y = functools.partial(adaptive_difference, axis=-2, step=dy, **kw)
    d_z = functools.partial(adaptive_difference, axis=-3, step=dz, coords=z, **kw)
    return xp.stack([d_x(v) - d_y(u), d_z(u) - d_x(w), d_y(w) - d_z(v)])

    adaptive_difference(f, *, mask=None, axis=-1, step=1.0, coords=None, derivative=1, accuracy=1, periodic=False)

    Mask-aware finite difference: the widest stencil that fits on valid data.

    Near a mask (land, a swath edge, a cloud) the centred stencil reads a masked cell and :func:difference loses the valid cells next to it. This keeps, per point, the first finite result among the central, forward and backward stencils at accuracy, then at each lower accuracy down to 1, so a narrow channel still gets a first-order estimate. Recovered cells carry one order less than the interior.

    NaN is its own support detector: a stencil that reads a masked cell returns NaN (0 * nan is nan), so the selection needs no mask arithmetic (xrtoolz docs/design/xrgrad-nan-masks.md, where it beat fill-then-differentiate by ~90x at the coast). Every candidate is computed, so the call costs 3 * accuracy stencils and runs under jax.jit. Masked cells are NaN in the result.

    Args: f: The field; any shape. mask: True drops the cell (the analysis mask convention: True = dropped), on f's shape; None drops only non-finite cells. axis: The axis to differentiate. step: The signed step between centres along axis. coords: The centres' positions along axis, for an unevenly spaced axis (see :func:difference); replaces step. derivative: The derivative order. accuracy: The highest accuracy order to try. periodic: The axis wraps.

    Returns: The derivative, f's shape, NaN where f is masked or no stencil fits on valid data.

    Example: >>> f = np.array([0.0, 1.0, 2.0, np.nan, 4.0, 5.0, 6.0]) # f = x >>> difference(f) array([ 1., 1., nan, nan, nan, 1., 1.]) >>> adaptive_difference(f) array([ 1., 1., 1., nan, 1., 1., 1.])

    def adaptive_difference(
    f: Float[np.ndarray, "*shape"],
    *,
    mask: Bool[np.ndarray, "*shape"] | None = None,
    axis: int = -1,
    step: Float[np.ndarray, "*shape"] | float = 1.0,
    coords: Float[np.ndarray, " n"] | None = None,
    derivative: int = 1,
    accuracy: int = 1,
    periodic: bool = False,
) -> Float[np.ndarray, "*shape"]:
    r"""Mask-aware finite difference: the widest stencil that fits on valid data.

    Near a mask (land, a swath edge, a cloud) the centred stencil reads a
    masked cell and :func:`difference` loses the valid cells next to it.
    This keeps, per point, the first finite result among the central,
    forward and backward stencils at ``accuracy``, then at each lower
    accuracy down to 1, so a narrow channel still gets a first-order
    estimate. Recovered cells carry one order less than the interior.

    NaN is its own support detector: a stencil that reads a masked cell
    returns NaN (``0 * nan`` is ``nan``), so the selection needs no mask
    arithmetic (xrtoolz ``docs/design/xrgrad-nan-masks.md``, where it beat
    fill-then-differentiate by ~90x at the coast). Every candidate is
    computed, so the call costs ``3 * accuracy`` stencils and runs under
    ``jax.jit``. Masked cells are ``NaN`` in the result.

    Args:
        f: The field; any shape.
        mask: ``True`` drops the cell (the analysis mask convention: ``True`` = dropped), on ``f``'s
            shape; ``None`` drops only non-finite cells.
        axis: The axis to differentiate.
        step: The signed step between centres along ``axis``.
        coords: The centres' positions along ``axis``, for an unevenly spaced
            axis (see :func:`difference`); replaces ``step``.
        derivative: The derivative order.
        accuracy: The highest accuracy order to try.
        periodic: The axis wraps.

    Returns:
        The derivative, ``f``'s shape, ``NaN`` where ``f`` is masked or no
        stencil fits on valid data.

    Example:
        >>> f = np.array([0.0, 1.0, 2.0, np.nan, 4.0, 5.0, 6.0])  # f = x
        >>> difference(f)
        array([ 1.,  1., nan, nan, nan,  1.,  1.])
        >>> adaptive_difference(f)
        array([ 1.,  1.,  1., nan,  1.,  1.,  1.])
    """
    xp = array_namespace(f)
    dropped = ~xp.isfinite(f)
    if mask is not None:
        dropped = dropped | mask
    field = xp.where(dropped, xp.nan, f)
    out = None
    for acc in range(accuracy, 0, -1):
        for method in METHODS:
            try:
                candidate = difference(
                    field,
                    axis=axis,
                    step=step,
                    coords=coords,
                    derivative=derivative,
                    accuracy=acc,
                    method=method,
                    periodic=periodic,
                )
            except ValueError:
                continue  # a stencil wider than the axis: try a narrower one
            out = (
                candidate if out is None else xp.where(xp.isfinite(out), out, candidate)
            )
    if out is None:
        raise ValueError(
            f"axis of length {f.shape[axis]} is too short for derivative {derivative}"
        )
    return xp.where(dropped, xp.nan, out)

    def adaptive_divergence_2d(
    u: Float[np.ndarray, "*batch h w"],
    v: Float[np.ndarray, "*batch h w"],
    dy: Float[np.ndarray, "h w"] | float = 1.0,

    *,
    mask: Bool[np.ndarray, "*batch h w"] | None = None,

    periodic_x: bool = False,
) -> Float[np.ndarray, "*batch h w"]:
    """Mask-aware divergence: :func:`adaptive_difference` for each derivative.

    Each term uses, per point, the widest stencil lying on valid data, so a
    linear field stays exact up to the coast. A cell is ``NaN`` where ``mask``
    drops it or either component is non-finite.

    Args:
        u: The component along columns.
        v: The component along rows.
        dy: The signed step between row centres.
        dx: The signed step between column centres.
        mask: ``True`` drops the cell; ``None`` drops only non-finite cells.
        accuracy: The highest accuracy order to try.
        periodic_x: The columns wrap.

    Returns:
        The divergence, ``u``'s shape.

    Example:
        >>> y, x = np.mgrid[0:5, 0:6].astype(float)
        >>> land = (y == 2) & (x == 3)
        >>> div = adaptive_divergence_2d(x, y, mask=land)
        >>> int(np.isnan(div).sum()), float(np.nanmax(np.abs(div - 2.0)))
        (1, 0.0)
    """
    d_u = adaptive_difference(
        u, mask=mask, axis=-1, step=dx, accuracy=accuracy, periodic=periodic_x
    )
    return d_u + adaptive_difference(v, mask=mask, axis=-2, step=dy, accuracy=accuracy)

def difference_coeffs(
    offsets: tuple[int, ...], derivative: int = 1

    r"""The coefficients of a stencil on ``offsets``, from the Taylor conditions.

    :math:`m = 0, \dots, K - 1` in float64.

        offsets: The stencil's integer offsets, distinct.
        derivative: The derivative order; fewer than ``derivative + 1``
            offsets cannot resolve it.

        ValueError: Too few offsets for the derivative.

    Examples:
        >>> difference_coeffs((-1, 0, 1), derivative=2)
        (1.0, -2.0, 1.0)
        >>> [round(c, 6) for c in difference_coeffs((0, 1, 2), derivative=1)]
        [-1.5, 2.0, -0.5]
    """
    n = len(offsets)
    if derivative >= n:
        raise ValueError(
            f"{n} offsets cannot resolve derivative {derivative}; need at least "
            f"{derivative + 1}"
        )
    powers = np.vander(np.asarray(offsets, dtype=np.float64), n, increasing=True).T
    rhs = np.zeros(n)
    rhs[derivative] = math.factorial(derivative)
    coeffs = np.linalg.solve(powers, rhs)
    # round off solver noise so exact stencils stay exact (0.5, not 0.4999…)
    return tuple(float(c) for c in np.round(coeffs, 12) + 0.0)

    def nonuniform_coeffs(dz: Any, derivative: int = 1) -> Any:
    r"""Stencil coefficients for points at arbitrary distances, on any backend.

    The non-uniform form of :func:`difference_coeffs`: for each point, the
    weights :math:`c_k` with :math:`f^{(n)}(z) \approx \sum_k c_k f(z + \delta_k)`
    solve the Taylor conditions :math:`\sum_k c_k \delta_k^m / m! = \delta_{mn}`
    for :math:`m = 0, \dots, K - 1`, so the stencil is exact for polynomials of
    degree :math:`K - 1`. Each row is solved in units of its own widest
    distance, which keeps the system well conditioned for coordinates in
    metres or pascals. Written on the array namespace, so traced distances
    (a ``TimeAxis`` under ``jax.jit``) work too.

    Args:
        dz: ``(..., K)`` signed distances :math:`\delta_k` from each point to
            its stencil's points; one of them is ``0``.
        derivative: The derivative order :math:`n`, below ``K``.

    Returns:
        ``(..., K)`` coefficients, in units of ``dz`` to the power ``-n``.

    Raises:
        ValueError: Too few points for the derivative.

    Examples:
        >>> nonuniform_coeffs(np.array([-1.0, 0.0, 1.0]))  # the uniform stencil
        array([-0.5,  0. ,  0.5])
        >>> # uneven levels at 0, 1 and 3 m: still exact for a quadratic
        >>> c = nonuniform_coeffs(np.array([-1.0, 0.0, 2.0]), derivative=2)
        >>> float(c @ np.array([-1.0, 0.0, 2.0]) ** 2)
        2.0
    """

    xp = array_namespace(dz)
    n_points = dz.shape[-1]
    if derivative >= n_points:
        raise ValueError(
            f"{n_points} points cannot resolve derivative {derivative}; need at "
            f"least {derivative + 1}"
        )
    scale = xp.max(xp.abs(dz), axis=-1, keepdims=True)
    unit = dz / scale
    # A[..., m, k] = δ_k^m / m!, in units of the row's widest distance
    powers = xp.stack([unit**m / math.factorial(m) for m in range(n_points)], axis=-2)
    rhs = xp.zeros((*dz.shape[:-1], n_points), dtype=unit.dtype)
    rhs = xp.where(xp.arange(n_points) == derivative, 1.0, rhs)
    coeffs = xp.linalg.solve(powers, rhs[..., None])[..., 0]
    return coeffs / scale**derivative

        is the same as interpolating the real and imaginary parts apart.

        concrete (NumPy, or a JAX array outside ``jit``): the orientation

        in ``values``' namespace and dtype.

        strictly monotonic, or ``method`` is unknown.

    >>> t = np.array([[290.0, 280.0, 250.0]])

    array([[285., 265.]])

        raise ValueError(f"unknown method {method!r}; expected 'linear' or 'nearest'")
    xp = array_namespace(values)
    arr = _as_inexact(xp, values)
    src_np = _concrete(source_coords)
    if src_np is None:
        raise ValueError("remap_axis needs concrete source_coords (not a tracer)")

    tgt = xp.asarray(target_coords, dtype=_real_dtype(xp, arr))

    def remap_axis_columns(
    values: Inexact[np.ndarray, "..."],
    *,
    axis: int,
    source_coords: Float[np.ndarray, "..."],
    target_coords: Float[np.ndarray, "tgt"],
    method: str = "linear",
    extrapolate: Extrapolate = "nan",
) -> Inexact[np.ndarray, "..."]:
    """Interpolate every column of ``values`` onto ``target_coords``.

    Unlike :func:`remap_axis`, ``source_coords`` has ``values``' shape: each
    column (each position along the other axes) carries its own levels, and
    each column must be strictly monotone along ``axis`` (ascending or
    descending; columns may differ). A column that broadcasts one 1-D
    coordinate gives :func:`remap_axis`'s result exactly.

    Parameters
    ----------
    values : array
        At least two levels along ``axis``; real or complex.
    axis : int
        The axis to remap.
    source_coords : array
        Source levels, ``values``' shape. May be traced: the orientation of
        each column is chosen with ``where``. The monotonicity check runs
        only on concrete levels.
    target_coords : array [tgt]
        Target levels; ``NaN`` targets give ``NaN``.
    method : {"linear", "nearest"}, optional
        Default ``"linear"``.
    extrapolate : {"nan", "nearest"}, optional
        Targets outside a column's own range: ``NaN`` (default), or that
        column's end value.

    Returns
    -------
    array
        ``values``' shape with ``axis`` of length ``len(target_coords)``.

    Raises
    ------
    ValueError
        If shapes disagree, ``target_coords`` is not 1-D, ``axis`` has fewer
        than two levels, a concrete column is not strictly monotone, or
        ``method`` / ``extrapolate`` is unknown.
    """
    if method not in _METHODS:
        raise ValueError(f"unknown method {method!r}; expected 'linear' or 'nearest'")
    if extrapolate not in _EXTRAPOLATE_MODES:
        raise ValueError(
            f"unknown extrapolate {extrapolate!r}; expected one of "
            f"{_EXTRAPOLATE_MODES!r}"
        )
    xp = array_namespace(values, source_coords)
    arr = _as_inexact(xp, values)
    src = xp.asarray(source_coords)
    tgt = xp.asarray(target_coords)
    if tgt.ndim != 1:
        raise ValueError("target_coords must be 1D")
    if tuple(src.shape) != tuple(arr.shape):
        raise ValueError(
            f"source_coords shape {tuple(src.shape)} must equal values shape "
            f"{tuple(arr.shape)}"
        )
    n_lev = arr.shape[axis]
    if n_lev < 2:
        raise ValueError(
            f"remap_axis_columns needs at least two levels along axis, got {n_lev}"
        )
    z = xp.moveaxis(src, axis, -1)
    f = xp.moveaxis(arr, axis, -1)
    z = xp.astype(z, _real_dtype(xp, arr))
    tgt = xp.astype(tgt, z.dtype)
    z_np = _concrete(z)
    if z_np is not None:
        steps = np.diff(z_np, axis=-1)
        bad = ~(np.all(steps > 0, axis=-1) | np.all(steps < 0, axis=-1))
        if bad.any():
            raise ValueError(
                "source_coords must be strictly monotone along axis in every column; "
                f"{int(bad.sum())} of {bad.size} columns are not"
            )
    # flip descending columns so every column is ascending
    descending = z[..., :1] > z[..., n_lev - 1 :]
    z = xp.where(descending, xp.flip(z, axis=-1), z)
    f = xp.where(descending, xp.flip(f, axis=-1), f)
    # one target at a time: a [columns, levels] temporary, not [columns,
    # targets, levels], as the NumPy kernel did
    out = xp.concat(
        [
            _interpolate(xp, f, z, tgt[k : k + 1], method, extrapolate)
            for k in range(tgt.shape[0])
        ],
        axis=-1,
    )
    return xp.moveaxis(out, -1, axis)
