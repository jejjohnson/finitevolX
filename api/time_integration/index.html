    Time Integration

    The generic time steppers live in integrix (integrix.march); finitevolX keeps the semi-Lagrangian scheme and the solve_ocean_pde wrapper.

    Deprecated finitevolX names

    finitevolx.RK3SSP, finitevolx.rk3_ssp_step and the other generic steppers below still work for one release and emit a DeprecationWarning. The diffrax solvers and the pure step functions are the integrix objects themselves, so import them from integrix.march. finitevolx.AB2Solver, finitevolx.LeapfrogRAFSolver and finitevolx.SplitExplicitRKSolver are adapters that keep the old init / step call shapes; their integrix replacements are integrix.march.AbstractSteppers run by integrix.march.run(stepper, rhs, t0, y0, dt, n_steps) with rhs(t, y, args).

    def euler_step(
    state: PyTree, rhs_fn: Callable[[PyTree], PyTree], dt: ScalarLike
) -> PyTree:
    """One forward-Euler step: yₙ₊₁ = yₙ + h f(yₙ).

    Args:
        state: The state yₙ, any pytree.
        rhs_fn: The autonomous right-hand side f(y).
        dt: The step h.

    Returns:
        The state yₙ₊₁.

    Examples:
        >>> import jax.numpy as jnp
        >>> bool(euler_step(jnp.array(1.0), lambda y: -y, 0.1) == 0.9)
        True
    """
    k1 = rhs_fn(state)
    return jax.tree.map(lambda y, f: y + dt * f, state, k1)

    def heun_step(
    state: PyTree, rhs_fn: Callable[[PyTree], PyTree], dt: ScalarLike
) -> PyTree:
    """One Heun (RK2) step, order 2, SSP with C = 1.

    k₁ = f(yₙ), k₂ = f(yₙ + h k₁), yₙ₊₁ = yₙ + (h/2)(k₁ + k₂).

    Args:
        state: The state yₙ, any pytree.
        rhs_fn: The autonomous right-hand side f(y).
        dt: The step h.

    Returns:
        The state yₙ₊₁.

    Examples:
        >>> import jax.numpy as jnp
        >>> bool(jnp.allclose(heun_step(jnp.array(1.0), lambda y: -y, 0.1), 0.905))
        True
    """
    k1 = rhs_fn(state)
    state_star = jax.tree.map(lambda y, f: y + dt * f, state, k1)
    k2 = rhs_fn(state_star)
    return jax.tree.map(lambda y, f1, f2: y + 0.5 * dt * (f1 + f2), state, k1, k2)

    def rk3_ssp_step(
    state: PyTree, rhs_fn: Callable[[PyTree], PyTree], dt: ScalarLike
) -> PyTree:
    """One third-order SSP Runge–Kutta step in Shu–Osher form, C = 1.

    y⁽¹⁾ = yₙ + h f(yₙ); y⁽²⁾ = ¾ yₙ + ¼ y⁽¹⁾ + ¼ h f(y⁽¹⁾);
    yₙ₊₁ = ⅓ yₙ + ⅔ y⁽²⁾ + ⅔ h f(y⁽²⁾).

    Args:
        state: The state yₙ, any pytree.
        rhs_fn: The autonomous right-hand side f(y).
        dt: The step h.

    Returns:
        The state yₙ₊₁.

    Examples:
        >>> import jax.numpy as jnp
        >>> y = rk3_ssp_step(jnp.array(1.0), lambda y: -y, 0.1)
        >>> bool(jnp.allclose(y, 1.0 - 0.1 + 0.1**2 / 2 - 0.1**3 / 6))
        True
    """
    k1 = rhs_fn(state)
    y1 = jax.tree.map(lambda y, f: y + dt * f, state, k1)

    k2 = rhs_fn(y1)
    y2 = jax.tree.map(
        lambda y, y1_, f: 0.75 * y + 0.25 * y1_ + 0.25 * dt * f,
        state,
        y1,
        k2,
    )

    k3 = rhs_fn(y2)
    return jax.tree.map(
        lambda y, y2_, f: (1.0 / 3.0) * y + (2.0 / 3.0) * y2_ + (2.0 / 3.0) * dt * f,
        state,
        y2,
        k3,
    )

    def rk4_step(
    state: PyTree, rhs_fn: Callable[[PyTree], PyTree], dt: ScalarLike
) -> PyTree:
    """One classical fourth-order Runge–Kutta step.

    k₁ = f(yₙ), k₂ = f(yₙ + (h/2) k₁), k₃ = f(yₙ + (h/2) k₂), k₄ = f(yₙ + h k₃),
    yₙ₊₁ = yₙ + (h/6)(k₁ + 2k₂ + 2k₃ + k₄).

    Args:
        state: The state yₙ, any pytree.
        rhs_fn: The autonomous right-hand side f(y).
        dt: The step h.

    Returns:
        The state yₙ₊₁.

    Examples:
        >>> import jax.numpy as jnp
        >>> y = rk4_step({"a": jnp.array(1.0)}, lambda y: {"a": -y["a"]}, 0.1)
        >>> bool(jnp.allclose(y["a"], jnp.exp(-0.1), rtol=1e-6))
        True
    """
    k1 = rhs_fn(state)
    y1 = jax.tree.map(lambda y, f: y + 0.5 * dt * f, state, k1)

    k2 = rhs_fn(y1)
    y2 = jax.tree.map(lambda y, f: y + 0.5 * dt * f, state, k2)

    k3 = rhs_fn(y2)
    y3 = jax.tree.map(lambda y, f: y + dt * f, state, k3)

    k4 = rhs_fn(y3)
    return jax.tree.map(
        lambda y, f1, f2, f3, f4: y + (dt / 6.0) * (f1 + 2.0 * f2 + 2.0 * f3 + f4),
        state,
        k1,
        k2,
        k3,
        k4,
    )

    def ab2_step(
    state: PyTree,
    rhs_fn: Callable[[PyTree], PyTree],
    dt: ScalarLike,
    rhs_nm1: PyTree,
) -> tuple[PyTree, PyTree, PyTree]:
    """One second-order Adams–Bashforth step.

    yₙ₊₁ = yₙ + (h/2)(3 fₙ − fₙ₋₁), one evaluation of f per step.

    Args:
        state: The state yₙ, any pytree.
        rhs_fn: The autonomous right-hand side f(y).
        dt: The step h.
        rhs_nm1: fₙ₋₁, the right-hand side one step back.

    Returns:
        ``(yₙ₊₁, fₙ, fₙ₋₁)``: thread fₙ as the next ``rhs_nm1``.

    Examples:
        >>> import jax.numpy as jnp
        >>> y, f_n, _ = ab2_step(jnp.array(1.0), lambda y: -y, 0.1, -1.0)
        >>> bool(jnp.allclose(y, 0.9)), bool(f_n == -1.0)
        (True, True)
    """
    rhs_n = rhs_fn(state)
    # yₙ₊₁ = yₙ + (h/2)(3 fₙ − fₙ₋₁)
    new_state = jax.tree.map(
        lambda y, fn, fnm1: y + (dt / 2.0) * (3.0 * fn - fnm1),
        state,
        rhs_n,
        rhs_nm1,
    )
    return new_state, rhs_n, rhs_nm1

    def ab3_step(
    state: PyTree,
    rhs_fn: Callable[[PyTree], PyTree],
    dt: ScalarLike,
    rhs_nm1: PyTree,
    rhs_nm2: PyTree,
) -> tuple[PyTree, PyTree, PyTree]:
    """One third-order Adams–Bashforth step.

    yₙ₊₁ = yₙ + (h/12)(23 fₙ − 16 fₙ₋₁ + 5 fₙ₋₂), one evaluation of f per
    step. The start must be at least second-order accurate to keep the
    global order 3: bootstrap y₁ and y₂ with `rk3_ssp_step`, as `AB3Solver`
    does; the crude start fₙ₋₂ = fₙ₋₁ = f(y₀) degrades AB3 to second order.

    Args:
        state: The state yₙ, any pytree.
        rhs_fn: The autonomous right-hand side f(y).
        dt: The step h.
        rhs_nm1: fₙ₋₁, the right-hand side one step back.
        rhs_nm2: fₙ₋₂, two steps back.

    Returns:
        ``(yₙ₊₁, fₙ, fₙ₋₁)``: thread fₙ as the next ``rhs_nm1`` and fₙ₋₁ as
        the next ``rhs_nm2``.

    Examples:
        >>> import jax.numpy as jnp
        >>> f = lambda y: -y
        >>> y, f_n, f_nm1 = ab3_step(jnp.array(1.0), f, 0.1, f(1.0), f(1.0))
        >>> bool(jnp.allclose(y, 0.9)), bool(f_n == -1.0)
        (True, True)
    """
    rhs_n = rhs_fn(state)
    # yₙ₊₁ = yₙ + (h/12)(23 fₙ − 16 fₙ₋₁ + 5 fₙ₋₂)
    new_state = jax.tree.map(
        lambda y, fn, fnm1, fnm2: (
            y + (dt / 12.0) * (23.0 * fn - 16.0 * fnm1 + 5.0 * fnm2)
        ),
        state,
        rhs_n,
        rhs_nm1,
        rhs_nm2,
    )
    return new_state, rhs_n, rhs_nm1

    def leapfrog_raf_step(
    state: PyTree,
    state_nm1: PyTree,
    rhs_fn: Callable[[PyTree], PyTree],
    dt: ScalarLike,
    alpha: float = 0.05,
) -> tuple[PyTree, PyTree]:
    """One leapfrog step with a Robert–Asselin filter on the middle level.

    ỹₙ₊₁ = yₙ₋₁ + 2h f(yₙ) and ȳₙ = yₙ + α (yₙ₋₁ − 2yₙ + ỹₙ₊₁). The filtered
    ȳₙ damps the computational mode of the three-level scheme. Second order
    for α = 0 and first order for α > 0 (the filter's O(αh) damping; the
    measured rate is about 1.1 at the default 0.05).

    Args:
        state: The state yₙ, any pytree.
        state_nm1: The (filtered) state yₙ₋₁.
        rhs_fn: The autonomous right-hand side f(y).
        dt: The step h.
        alpha: The filter coefficient α, typically 0.01 to 0.1.

    Returns:
        ``(yₙ₊₁, ȳₙ)``: yₙ₊₁ is the next state and ȳₙ the next ``state_nm1``.

    Examples:
        >>> import jax.numpy as jnp
        >>> y_next, filtered = leapfrog_raf_step(1.0, 1.0, lambda y: 0.0 * y, 0.1)
        >>> bool(y_next == 1.0), bool(filtered == 1.0)
        (True, True)
    """
    rhs_n = rhs_fn(state)

    # ỹₙ₊₁ = yₙ₋₁ + 2h f(yₙ)
    y_next = jax.tree.map(
        lambda ynm1, fn: ynm1 + 2.0 * dt * fn,
        state_nm1,
        rhs_n,
    )

    # ȳₙ = yₙ + α (yₙ₋₁ − 2yₙ + ỹₙ₊₁)
    state_filtered = jax.tree.map(
        lambda ynm1, yn, ynp1: yn + alpha * (ynm1 - 2.0 * yn + ynp1),
        state_nm1,
        state,
        y_next,
    )

    return y_next, state_filtered

    state: PyTree,
    rhs_explicit: Callable[[PyTree], PyTree],

    Second order in the explicit part, the implicit part and the coupled
    split (the implicit row (1 − 2γ, γ) gives ∑ᵢ bᵢ cᵢ = ½).

    Algorithm:
        1. Y₁ = yₙ + γh f_I(Y₁): ``implicit_solve(yₙ, γh)``.
        2. Y₂* = yₙ + h f_E(Y₁) + (1 − 2γ) h f_I(Y₁), then
           Y₂ = Y₂* + γh f_I(Y₂): ``implicit_solve(Y₂*, γh)``.
        3. yₙ₊₁ = yₙ + (h/2)(f_E(Y₁) + f_E(Y₂)) + (h/2)(f_I(Y₁) + f_I(Y₂)).

    Args:
        state: The state yₙ, any pytree.
        rhs_explicit: The non-stiff right-hand side f_E(y).
        rhs_implicit: The stiff right-hand side f_I(y).
        implicit_solve: Solves Y − γh f_I(Y) = r for Y, given (r, γh); for
            vertical diffusion, a tridiagonal solve along columns.
        dt: The step h.

    Returns:
        The state yₙ₊₁.

    Examples:
        >>> import jax.numpy as jnp
        >>> y = imex_ssp2_step(
        ...     jnp.array(1.0),
        ...     lambda y: -y,
        ...     lambda y: -10.0 * y,
        ...     lambda r, gamma_dt: r / (1.0 + 10.0 * gamma_dt),
        ...     0.01,
        ... )
        >>> bool(jnp.allclose(y, jnp.exp(-0.11), rtol=1e-4))
        True
    """
    gamma_dt = _GAMMA * dt

    # Stage 1: Y₁ = yₙ + γh f_I(Y₁)
    y1 = implicit_solve(state, gamma_dt)
    fe_1 = rhs_explicit(y1)
    fi_1 = rhs_implicit(y1)

    # Stage 2: Y₂* = yₙ + h f_E(Y₁) + (1 − 2γ) h f_I(Y₁); Y₂ = Y₂* + γh f_I(Y₂)
    y2_star = jax.tree.map(
        lambda y, fe, fi: y + dt * fe + (1.0 - 2.0 * _GAMMA) * dt * fi,
        state,
        fe_1,
        fi_1,
    )
    y2 = implicit_solve(y2_star, gamma_dt)
    fe_2 = rhs_explicit(y2)
    fi_2 = rhs_implicit(y2)

    # yₙ₊₁ = yₙ + (h/2)(f_E(Y₁) + f_E(Y₂)) + (h/2)(f_I(Y₁) + f_I(Y₂))
    return jax.tree.map(
        lambda y, fe1, fe2, fi1, fi2: (
            y + 0.5 * dt * (fe1 + fe2) + 0.5 * dt * (fi1 + fi2)
        ),
        state,
        fe_1,
        fe_2,
        fi_1,
        fi_2,
    )

    state_3d: PyTree,

    The fast subcycle is neutrally stable for the oscillatory gravity-wave
    mode (RK4 amplification 1 − (ωh)⁶/144 per substep); the slow step is first
    order in h. The returned 2-D state is the end-of-cycle y_M, which stays on
    time; the slow step used the mean ⟨y⟩, and making the 3-D state consistent
    with it (replacing the depth mean of the 3-D velocity by the averaged
    transport, say) is ``couple_2d_to_3d``'s job.

        2. Subcycle the 2-D state with M RK4 steps of h/M on
           y′ = ``rhs_2d(t, y, state_3d)`` + F, and form its trapezoidal
           mean ⟨y⟩ over the cycle.
        3. One forward-Euler step of the 3-D state:
           y₃ + h ``rhs_3d(state_3d, ⟨y⟩)``.
        4. ``couple_2d_to_3d(new state_3d, ⟨y⟩)``.

        state_2d: The fast (barotropic) state.

            the slow step.
        dt_slow: The slow step h.
        n_substeps: The number of fast substeps M ≥ 1 per slow step.
        slow_forcing: ``slow_forcing(state_3d)``, the baroclinic forcing of the
            barotropic mode, evaluated once per slow step and added to every
            fast tendency.

        >>> import jax.numpy as jnp
        >>> slow, fast = split_explicit_step(
        ...     jnp.array(1.0),
        ...     jnp.array(1.0),
        ...     lambda y3, mean_2d: -mean_2d,

    y_3d_new = couple_2d_to_3d(y_3d_new, y_2d_avg)

    return y_3d_new, y_2d_end

        Grid spacing in x and y (m).

        Timestep (s).

        Currently JAX only supports ``order <= 1``.  Default 1.

    Array[Ny, Nx]

        raise ValueError(f"bc must be 'periodic' or 'edge', got {bc!r}")
    if interp_order not in {0, 1}:
        raise ValueError(
            f"interp_order must be 0 or 1 (JAX limitation), got {interp_order}"
        )

    ny, nx = field.shape

    # Target grid coordinates (in index space)
    y_coords, x_coords = jnp.meshgrid(
        jnp.arange(ny, dtype=field.dtype),
        jnp.arange(nx, dtype=field.dtype),
        indexing="ij",
    )

    # Departure points in index space
    x_dep = x_coords - u * dt / dx
    y_dep = y_coords - v * dt / dy

    mode = "wrap" if bc == "periodic" else "nearest"

    return jax.scipy.ndimage.map_coordinates(
        field,
        [y_dep, x_dep],
        order=interp_order,
        mode=mode,
    )

    class AB3Solver(dfx.AbstractSolver):
    """Third-order Adams–Bashforth as a diffrax solver, with an SSP-RK3 start.

    yₙ₊₁ = yₙ + (h/12)(23 fₙ − 16 fₙ₋₁ + 5 fₙ₋₂), one evaluation of F per
    step. The two-level history lives in the diffrax solver state, and the
    first two steps are SSP-RK3 steps (a third-order bootstrap).

    Fixed, uniform steps only (``dt0=``, or ``StepTo`` with equal spacing):
    the weights assume a constant step and there is no error estimate, so an
    adaptive controller raises. Order 3; dense output is linear.

    Examples:
        >>> import diffrax as dfx
        >>> import jax.numpy as jnp
        >>> solution = dfx.diffeqsolve(
        ...     dfx.ODETerm(lambda t, y, args: -y),
        ...     AB3Solver(),
        ...     t0=0.0,
        ...     t1=1.0,
        ...     dt0=0.01,
        ...     y0=jnp.array(1.0),
        ... )
        >>> bool(jnp.allclose(solution.ys[0], jnp.exp(-1.0), rtol=1e-5))
        True
    """

    term_structure: ClassVar[Any] = dfx.AbstractTerm
    interpolation_cls: ClassVar[Any] = dfx.LocalLinearInterpolation

    def order(self, terms):
        return 3

    def init(self, terms, t0, t1, y0, args):
        # (fₙ₋₁, fₙ₋₂, number of steps taken)
        f0 = terms.vf(t0, y0, args)
        zeros = jax.tree.map(jnp.zeros_like, f0)
        return zeros, zeros, jnp.asarray(0, dtype=jnp.int32)

    def step(self, terms, t0, t1, y0, args, solver_state, made_jump):
        del made_jump
        f_nm1, f_nm2, n_taken = solver_state
        dt = t1 - t0
        f_n = terms.vf(t0, y0, args)

        def _rk3(_):
            # SSP-RK3 bootstrap (reuses fₙ as its first stage)
            y1 = jax.tree.map(lambda y, f: y + dt * f, y0, f_n)
            k2 = terms.vf(t1, y1, args)
            y2 = jax.tree.map(
                lambda y, a, b: 0.75 * y + 0.25 * (a + dt * b), y0, y1, k2
            )
            k3 = terms.vf(t0 + 0.5 * dt, y2, args)
            return jax.tree.map(
                lambda y, a, b: y / 3.0 + (2.0 / 3.0) * (a + dt * b), y0, y2, k3
            )

        def _ab3(_):
            # fₙ is already evaluated, so hand ab3_step a constant right-hand side
            y_next, _, _ = ab3_step(y0, lambda _y: f_n, dt, f_nm1, f_nm2)
            return y_next

        y1 = jax.lax.cond(n_taken < 2, _rk3, _ab3, None)
        dense_info = dict(y0=y0, y1=y1)
        new_state = (f_n, f_nm1, n_taken + 1)
        return y1, None, dense_info, new_state, RESULTS.successful

    def func(self, terms, t0, y0, args):
        return terms.vf(t0, y0, args)

    t0: ScalarLike,
    y0: PyTree,
    dt: ScalarLike,
    n_steps: int,
    *,
    args: Any = None,
    saveat: dfx.SaveAt | None = None,
    adjoint: Any = None,
) -> PyTree:
    """March y₀ through ``n_steps`` constant steps of h from t₀.

    yₙ₊₁ = Φ(yₙ, tₙ, h) for n = 0, …, N − 1, with tₙ = t₀ + n h.

    Algorithm:
        1. A diffrax solver goes to `diffrax.diffeqsolve` on the step times
           t₀ + n h (`diffrax.StepTo`), so it takes exactly N steps.
        2. An `AbstractStepper` takes its first step with ``init`` and the
           other N − 1 with ``step`` inside `jax.lax.scan`, carrying the
           state and its history.

    Args:
        stepper: An `AbstractStepper`, or a diffrax solver (the steppers of
            `integrix.march`, or diffrax's own).
        rhs: The vector field ``rhs(t, y, args)``. A diffrax solver may also
            take a diffrax term (a `diffrax.MultiTerm` for IMEX, say); a
            split-explicit stepper takes its `SplitExplicitTerms`.
        t0: The start time t₀.
        y0: The initial state y₀, any pytree.
        dt: The step h.
        n_steps: The number of steps N ≥ 1, a Python integer.
        args: Passed to the vector field.
        saveat: ``None`` for the final state y_N, or ``SaveAt(steps=True)``
            for every state y₁, …, y_N, stacked on a new leading axis.
        adjoint: How reverse mode runs back through the march. ``None`` is
            plain reverse mode (every state stored for a stepper; diffrax's
            default for a solver). `integrix.adjoint.Checkpointed` stores
            every k-th state and recomputes between (for a diffrax solver,
            ``RecursiveCheckpointAdjoint(checkpoints=⌈N / k⌉)``).
            `integrix.adjoint.Truncated` cuts the gradient through the state
            before the last k steps, for a stepper only. A diffrax solver
            also takes a diffrax adjoint or a pipekit-cycle spec
            (`integrix.adjoint.to_diffrax_adjoint`). Forward values never
            depend on the adjoint.

    Returns:
        y_N, or the stacked y₁, …, y_N.

    Raises:
        ValidationError: If ``n_steps`` is not a positive integer, ``saveat``
            asks for anything but the final state or every step, ``adjoint``
            does not apply to the stepper (`Truncated` for a diffrax solve, a
            diffrax adjoint for an `AbstractStepper`), or ``stepper`` is
            neither kind.

    Examples:
        >>> import jax.numpy as jnp
        >>> from integrix.march import RK4Classic
        >>> y = run(
        ...     RK4Classic(), lambda t, y, args: -y, 0.0, jnp.array(1.0), 0.1, 10
        ... )
        >>> bool(jnp.allclose(y, jnp.exp(-1.0), rtol=1e-6))
        True
        >>> from integrix.march import AB2Solver
        >>> every = run(
        ...     AB2Solver(),
        ...     lambda t, y, args: -y,
        ...     0.0,
        ...     jnp.ones(3),
        ...     0.1,
        ...     5,
        ...     saveat=dfx.SaveAt(steps=True),
        ... )
        >>> every.shape
        (5, 3)
    """

        n_steps = operator.index(n_steps)
    except TypeError as err:
        raise ValidationError(
            f"'n_steps' must be a Python integer, got {type(n_steps).__name__}"
        ) from err
    if n_steps < 1:
        raise ValidationError(f"'n_steps' must be at least 1, got {n_steps}")
    save_steps = _save_steps(saveat)

    if isinstance(stepper, dfx.AbstractSolver):
        return _run_diffrax(
            stepper, rhs, t0, y0, dt, n_steps, args, save_steps, adjoint
        )
    if not isinstance(stepper, AbstractStepper):
        raise ValidationError(
            f"'stepper' must be an AbstractStepper or a diffrax solver, got "
            f"{type(stepper).__name__}"
        )
    scan = _loop_scan(adjoint, n_steps)

    time_dtype = state_time_dtype(y0)
    start = jnp.asarray(t0, dtype=time_dtype)
    step_size = jnp.asarray(dt, dtype=time_dtype)

    state, y1 = stepper.init(rhs, start, y0, step_size, args)
    # y₁ as saved: its own step's gradient survives any truncation
    y1_saved = y1
    if isinstance(adjoint, Truncated) and adjoint.k < n_steps:
        # the first step lies before the last k, so the carry it hands on is cut
        y1, state = jax.lax.stop_gradient((y1, state))

    def body(carry, index):
        y, history = carry
        # tₙ = t₀ + n h
        t = start + index.astype(time_dtype) * step_size
        y_next, history = stepper.step(rhs, t, y, step_size, history, args)
        return (y_next, history), (y_next if save_steps else None)

    indices = jnp.arange(1, n_steps)
    (y_final, _), later = scan(body, (y1, state), indices)
    if not save_steps:
        return y_final
    return jax.tree.map(
        lambda first, rest: jnp.concatenate([first[None], rest]), y1_saved, later
    )

    class SemiLagrangianSolver(dfx.AbstractSolver):
    """Semi-Lagrangian advection solver for diffrax.

    Traces characteristic curves backward in time and interpolates the old
    field at departure points.  Unconditionally stable (CFL > 1 allowed).

    The term's ``vf(t, y, args)`` must return ``(u, v)`` velocity components
    in **grid index units per second** (i.e. physical velocity divided by
    grid spacing); wrap the velocity function in :class:`SemiLagrangianTerm`
    to use the solver with ``diffeqsolve``.  The departure point is found
    with one backward Euler step along a velocity frozen at ``t0``, so the
    scheme is first order in time (:meth:`order` is 1); there is no error
    estimate, so it runs with fixed steps only.  Dense output (``SaveAt(ts=...)``)
    is linear between steps.

    Parameters
    ----------
    interpolation_order : int
        0 = nearest-neighbour, 1 = linear (diffusive, monotone).
        JAX currently only supports ``order <= 1``.
    """

    term_structure: ClassVar[Any] = dfx.AbstractTerm
    interpolation_cls: ClassVar[Any] = dfx.LocalLinearInterpolation
    interpolation_order: int = eqx.field(static=True, default=1)

    def order(self, terms):
        # first-order backward trajectory; the spatial interpolation order
        # (interpolation_order) is a separate, spatial accuracy
        return 1

    def init(self, terms, t0, t1, y0, args):
        return None

    def step(self, terms, t0, t1, y0, args, solver_state, made_jump):
        del solver_state, made_jump
        dt = t1 - t0

        # Velocity in grid-index units/second
        u, v = terms.vf(t0, y0, args)

        ny, nx = y0.shape
        y_coords, x_coords = jnp.meshgrid(
            jnp.arange(ny, dtype=y0.dtype),
            jnp.arange(nx, dtype=y0.dtype),
            indexing="ij",
        )

        x_dep = x_coords - u * dt
        y_dep = y_coords - v * dt

        y1 = jax.scipy.ndimage.map_coordinates(
            y0, [y_dep, x_dep], order=self.interpolation_order, mode="wrap"
        )

        dense_info = dict(y0=y0, y1=y1)
        return y1, None, dense_info, None, RESULTS.successful

    def func(self, terms, t0, y0, args):
        return terms.vf(t0, y0, args)

    def step(
    self,
    rhs_slow: Callable,
    rhs_fast: Callable,
    t: float,
    y_3d,
    y_2d,
    dt_slow: float,
    slow_forcing: Callable | None = None,
    couple_2d_to_3d: Callable | None = None,
):
    """Take one split-explicit step.

    Parameters
    ----------
    rhs_slow : Callable[[float, PyTree, PyTree], PyTree]
        Slow RHS: ``rhs_slow(t, y_3d, y_2d_avg) -> tendency_3d``.
    rhs_fast : Callable[[float, PyTree, PyTree], PyTree]
        Fast RHS: ``rhs_fast(t_sub, y_2d, y_3d) -> tendency_2d``, with
        ``t_sub`` the absolute time of the RK4 stage.
    t : float
        Current time.
    y_3d, y_2d : PyTree
        3D (slow) and 2D (fast) states.
    dt_slow : float
        Slow timestep.
    slow_forcing : Callable[[PyTree], PyTree] or None, optional
        ``slow_forcing(y_3d) -> forcing_2d``, evaluated once and added to
        every fast tendency.
    couple_2d_to_3d : Callable[[PyTree, PyTree], PyTree] or None, optional
        ``couple_2d_to_3d(new_y_3d, y_2d_avg) -> corrected_y_3d``,
        applied after the slow step.

    Returns
    -------
    tuple[PyTree, PyTree]
        ``(new_y_3d, new_y_2d)``; ``new_y_2d`` is the end-of-cycle fast
        state at ``t + dt_slow``.
    """
    terms = ixm.SplitExplicitTerms(
        slow=lambda t_, y3, mean_2d, args: rhs_slow(t_, y3, mean_2d),
        fast=lambda t_, y2, y3, args: rhs_fast(t_, y2, y3),
        slow_forcing=(
            None
            if slow_forcing is None
            else (lambda t_, y3, args: slow_forcing(y3))
        ),
        couple=(
            None
            if couple_2d_to_3d is None
            else (lambda y3, mean_2d, args: couple_2d_to_3d(y3, mean_2d))
        ),
    )
    stepper = ixm.SplitExplicitRKSolver(n_substeps=self.n_substeps)
    (new_3d, new_2d), _ = stepper.step(terms, t, (y_3d, y_2d), dt_slow, None, None)
    return new_3d, new_2d

    def solve_ocean_pde(
    rhs_fn: Callable,
    solver: dfx.AbstractSolver,
    y0: PyTree,
    t0: float,
    t1: float,
    dt0: float,
    saveat: dfx.SaveAt | None = None,
    bc_fn: Callable | None = None,
    args: PyTree = None,
    **kwargs: Any,
) -> dfx.Solution:
    """Integrate an ocean PDE using diffrax.

    Parameters
    ----------
    rhs_fn : Callable
        Right-hand side ``rhs_fn(t, y, args) -> dy/dt``.
    solver : diffrax.AbstractSolver
        Time integration scheme (e.g. ``integrix.march.RK3SSP()``,
        ``integrix.march.RK4Classic()``).
    y0 : PyTree
        Initial condition.
    t0, t1 : float
        Start and end times.
    dt0 : float
        Initial (or fixed) timestep.
    saveat : diffrax.SaveAt, optional
        Output saving specification.  Defaults to saving the final state.
    bc_fn : Callable, optional
        Boundary condition function ``bc_fn(dydt) -> dydt_corrected`` applied
        to the tendency after each RHS evaluation.
    args : PyTree, optional
        Static arguments forwarded to ``rhs_fn``.
    **kwargs
        Forwarded to :func:`diffrax.diffeqsolve` (``stepsize_controller``,
        ``adjoint``, ``max_steps``, ``throw``, ``event``, ...).  For a
        fixed-step run (no ``stepsize_controller``) with concrete times and
        no ``max_steps``, ``max_steps`` defaults to
        ``ceil((t1 - t0) / dt0) + 16``: diffrax's own default (4096) stops a
        long fixed-step run early.  Otherwise diffrax's default applies.

    Returns
    -------
    diffrax.Solution
        Solution object containing the saved states.
    """
    if bc_fn is not None:

        def rhs_with_bc(t, y, args_):
            dydt = rhs_fn(t, y, args_)
            return bc_fn(dydt)

        term = dfx.ODETerm(rhs_with_bc)
    else:
        term = dfx.ODETerm(rhs_fn)

    if "max_steps" not in kwargs and "stepsize_controller" not in kwargs:
        # fixed steps: exactly ceil((t1 - t0) / dt0) of them
        try:
            n_steps = math.ceil((float(t1) - float(t0)) / float(dt0))
        except (TypeError, jax.errors.ConcretizationTypeError):
            pass  # traced times: keep diffrax's default
        else:
            kwargs["max_steps"] = max(n_steps, 0) + 16

    return dfx.diffeqsolve(
        term,
        solver,
        t0=t0,
        t1=t1,
        dt0=dt0,
        y0=y0,
        args=args,
        saveat=saveat if saveat is not None else dfx.SaveAt(t1=True),
        **kwargs,
    )
